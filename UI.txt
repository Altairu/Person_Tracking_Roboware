    <script>
        let websocket;
        let currentMode = 0;
        let stop = 0;
        let lx = 100, ly = 100, rx = 100, ry = 100;
        const reconnectInterval = 5000;

        // --- WebSocket 接続 ---
        function connectWebSocket() {
            websocket = new WebSocket("ws://10.53.255.212:8080/ws");

            websocket.onopen = function () {
                document.getElementById('ws-status').innerText = "WebSocket: 接続済み";
                document.getElementById('ws-status').className = "ws-status connected";
                // 起動時に現在のパラメータを取得
                websocket.send(JSON.stringify({ type: 'get_mdd_params' }));
            };

            websocket.onclose = function () {
                document.getElementById('ws-status').innerText = "WebSocket: 切断中";
                document.getElementById('ws-status').className = "ws-status disconnected";
                setTimeout(connectWebSocket, reconnectInterval);
            };

            websocket.onerror = function (error) {
                console.error("WebSocket error:", error);
            };

            websocket.onmessage = function (event) {
                try {
                    const data = JSON.parse(event.data);
                    if (data.type === 'mdd_params_ack') {
                        applyParamsToUI(data.params);
                        updateCanStatus(data.can_connected, data.can_port);
                        return;
                    }
                } catch (e) {
                    // JSON でなければ無視 (既存の CSV 応答など)
                }
            };
        }

        // --- CAN 状態表示 ---
        function updateCanStatus(connected, port) {
            const dot = document.getElementById('can-dot');
            const text = document.getElementById('can-status-text');
            if (connected) {
                dot.className = 'can-dot connected';
                text.innerText = 'CAN: 接続済み (' + port + ')';
            } else {
                dot.className = 'can-dot disconnected';
                text.innerText = 'CAN: 未接続';
            }
        }

        // --- モード切り替え ---
        function toggleMode() {
            const modeButton = document.getElementById("modeToggle");
            currentMode = currentMode === 0 ? 1 : 0;
            if (currentMode === 1) {
                modeButton.innerText = "Mode: Follow";
                modeButton.classList.add("active");
            } else {
                modeButton.innerText = "Mode: Control";
                modeButton.classList.remove("active");
            }
            sendCommand();
        }

        // --- ゲームパッド ---
        function updateGamepadValues() {
            const gamepads = navigator.getGamepads();
            const gamepad = gamepads[0];
            if (gamepad) {
                lx = gamepad.axes[0] * 100 + 100;
                ly = gamepad.axes[1] * -100 + 100;
                rx = gamepad.axes[2] * 100 + 100;
                ry = gamepad.axes[3] * -100 + 100;
                sendCommand();
            }
        }

        function sendCommand() {
            if (websocket && websocket.readyState === WebSocket.OPEN) {
                const command = [currentMode, rx, ry, lx, ly, stop];
                websocket.send(command.join(","));
            }
        }

        // --- MDD パラメータ ---
        function collectParams() {
            return {
                ch1: {
                    p:           parseFloat(document.getElementById('ch1-p').value)   || 0,
                    i:           parseFloat(document.getElementById('ch1-i').value)   || 0,
                    d:           parseFloat(document.getElementById('ch1-d').value)   || 0,
                    direction:   parseInt(document.getElementById('ch1-dir').value)   || 1,
                    diameter_mm: parseFloat(document.getElementById('ch1-dia').value) || 100,
                },
                ch2: {
                    p:           parseFloat(document.getElementById('ch2-p').value)   || 0,
                    i:           parseFloat(document.getElementById('ch2-i').value)   || 0,
                    d:           parseFloat(document.getElementById('ch2-d').value)   || 0,
                    direction:   parseInt(document.getElementById('ch2-dir').value)   || 1,
                    diameter_mm: parseFloat(document.getElementById('ch2-dia').value) || 100,
                }
            };
        }

        function applyParamsToUI(params) {
            if (!params) return;
            const ch1 = params.ch1 || {};
            const ch2 = params.ch2 || {};
            document.getElementById('ch1-p').value   = ch1.p           ?? 1.0;
            document.getElementById('ch1-i').value   = ch1.i           ?? 0.0;
            document.getElementById('ch1-d').value   = ch1.d           ?? 0.0;
            document.getElementById('ch1-dir').value = ch1.direction   ?? 1;
            document.getElementById('ch1-dia').value = ch1.diameter_mm ?? 100;
            document.getElementById('ch2-p').value   = ch2.p           ?? 1.0;
            document.getElementById('ch2-i').value   = ch2.i           ?? 0.0;
            document.getElementById('ch2-d').value   = ch2.d           ?? 0.0;
            document.getElementById('ch2-dir').value = ch2.direction   ?? 1;
            document.getElementById('ch2-dia').value = ch2.diameter_mm ?? 100;
        }

        function saveParams() {
            if (!websocket || websocket.readyState !== WebSocket.OPEN) {
                showFeedback('WebSocket が接続されていません。', true);
                return;
            }
            const params = collectParams();
            websocket.send(JSON.stringify({ type: 'mdd_params', params: params }));
            showFeedback('送信しました。応答待ち...');
        }

        function reloadParams() {
            if (!websocket || websocket.readyState !== WebSocket.OPEN) {
                showFeedback('WebSocket が接続されていません。', true);
                return;
            }
            websocket.send(JSON.stringify({ type: 'get_mdd_params' }));
            showFeedback('読み込み中...');
        }

        function showFeedback(msg, isError = false) {
            const el = document.getElementById('save-feedback');
            el.innerText = msg;
            el.className = 'save-feedback' + (isError ? ' error' : '');
            if (!isError) {
                setTimeout(() => { el.innerText = ''; }, 3000);
            }
        }

        function setupGamepad() {
            setInterval(updateGamepadValues, 100);
        }

        document.addEventListener('contextmenu', event => event.preventDefault());
        connectWebSocket();
        setupGamepad();
    </script>
